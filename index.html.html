                console.error('Error saving state:', e);

            }

        }

 

        // --- 2. المؤثرات الصوتية باستخدام Tone.js ---

        let synth = null;

        function initAudio() {

            if (!synth && typeof Tone !== 'undefined') {

                synth = new Tone.PolySynth(Tone.Synth).toDestination();

                synth.volume.value = -8;

            }

        }

 

        function playCelebrationSound(isMilestone = false, isComplete = false) {

            if (!appState.soundEnabled) return;

            try {

                initAudio();

                if (Tone.context.state !== 'running') {

                    Tone.start();

                }

                const now = Tone.now();

                if (isComplete) {

                    // نغمة انتصار كاملة عند الوصول لـ 50

                    synth.triggerAttackRelease("C5", "16n", now);

                    synth.triggerAttackRelease("E5", "16n", now + 0.1);

                    synth.triggerAttackRelease("G5", "16n", now + 0.2);

                    synth.triggerAttackRelease("C6", "4n", now + 0.3);

                } else if (isMilestone) {

                    // نغمة ترقية مستوى

                    synth.triggerAttackRelease("E5", "16n", now);

                    synth.triggerAttackRelease("G5", "16n", now + 0.12);

                    synth.triggerAttackRelease("B5", "8n", now + 0.24);

                } else {

                    // نغمة حجز عادي مبهجة

                    synth.triggerAttackRelease("G5", "32n", now);

                    synth.triggerAttackRelease("C6", "16n", now + 0.08);

                }

            } catch (e) {

                console.warn('Audio playback skipped:', e);

            }

        }

 

        function toggleSound() {

            appState.soundEnabled = !appState.soundEnabled;

            saveState();

            updateSoundUI();

            if (appState.soundEnabled) {

                playCelebrationSound(false, false);

            }

        }

 

        function updateSoundUI() {

            const icon = document.getElementById('soundIcon');

            const text = document.getElementById('soundText');

            if (appState.soundEnabled) {

                icon.className = 'fa-solid fa-volume-high text-amber-300';

                text.textContent = 'الصوت مفعل';

            } else {

                icon.className = 'fa-solid fa-volume-xmark text-slate-300';

                text.textContent = 'الصوت مكتوم';

            }

        }

 

        // --- 3. الرسائل التشجيعية والأوسمة (Gamification Logic) ---

        function getMilestoneInfo(count, target) {

            const ratio = count / target;

            if (count >= target) {

                return { title: 'بطل المبيعات الأسطوري', icon: '👑', badgeColor: 'bg-amber-100 text-amber-800 border-amber-300', level: 4 };

            } else if (ratio >= 0.8) { // 40+

                return { title: 'المرحلة الأخيرة (قناص)', icon: '🔥', badgeColor: 'bg-rose-100 text-rose-800 border-rose-300', level: 3 };

            } else if (ratio >= 0.5) { // 25+

                return { title: 'كابتن نص الطريق', icon: '🏅', badgeColor: 'bg-sky-100 text-sky-800 border-sky-300', level: 2 };

            } else if (ratio >= 0.2) { // 10+

                return { title: 'انطلاقة صاروخية', icon: '🚀', badgeColor: 'bg-emerald-100 text-emerald-800 border-emerald-300', level: 1 };

            } else {

                return { title: 'تجهيز للإقلاع', icon: '🛫', badgeColor: 'bg-slate-100 text-slate-600 border-slate-200', level: 0 };

            }

        }

 

        function getEncouragingMessage(rep) {

            const { name, count, target } = rep;

            const remaining = Math.max(0, target - count);

            const ratio = count / target;

 

            if (count === target) {

                return { emoji: '🏆', msg: `مبرووووك يا ${name}! قفلت الـ ${target} عميل بنجاح ساحق! أنت أسطورة المبيعات 🎉` };

            }

            if (count > target) {

                return { emoji: '🌟', msg: `عديت التارجت يا ${name}! (${count}/${target}) أداء خيالي ومكافأة مستحقة بجدارة! 🚀` };

            }

            if (remaining === 1) {

                return { emoji: '😱', msg: `عميل واحد فقط يا ${name} وتقفل التارجت! يلا الضربة القاضية 🔥` };

            }

            if (remaining <= 5) {

                return { emoji: '⚡', msg: `فاضلك ${remaining} عملاء بس يا ${name}! شايفين خط النهاية قدامك، وحش! 💪` };

            }

            if (ratio >= 0.8) {

                return { emoji: '🔥', msg: `دخلت منطقة الخطر يا ${name}! وصلت لـ ${count}/${target}.. الـ ${target} بتقرب جداً!` };

            }

            if (count === Math.floor(target / 2)) {

                return { emoji: '🎯', msg: `عاش يا ${name}! وصلت لنص الطريق بالظبط (${count}/${target})! كمل بنفس القوة ✈️` };

            }

            if (ratio >= 0.5) {

                return { emoji: '✈️', msg: `طيارتك في السحاب يا ${name}! عديت النص (${count}/${target}) ولسه القادم أقوى!` };

            }

            if (ratio >= 0.2) {

                return { emoji: '🚀', msg: `انطلاقة ولا أروع يا ${name}! (${count}/${target}) طيارتك أقلعت خلاص واستعد للتحليق!` };

            }

            const starters = [

                `بداية قوية يا ${name}! (${count}/${target}) كل رحلة ألف ميل بتبدأ بحجز 💪`,

                `عاش يا بطل! حجز جديد لـ ${name} (${count}/${target}).. استمر يا نجم ⭐`,

                `تسلم إيدك يا ${name}! العداد بيتحرك (${count}/${target}) والمركز الأول بيناديك 🏆`

            ];

            return { emoji: '🎉', msg: starters[count % starters.length] };

        }

 

        // --- 4. التفاعلات الرئيسية (إضافة / تراجع) ---

        function addBooking(repId) {

            const rep = appState.reps.find(r => r.id === repId);

            if (!rep) return;

 

            const prevMilestone = getMilestoneInfo(rep.count, rep.target);

            rep.count += 1;

            const newMilestone = getMilestoneInfo(rep.count, rep.target);

 

            const isMilestoneUp = newMilestone.level > prevMilestone.level;

            const isTargetReached = rep.count === rep.target;

 

            // تحديث البانر التشجيعي

            const cheer = getEncouragingMessage(rep);

            const banner = document.getElementById('cheerBanner');

            document.getElementById('cheerEmoji').textContent = cheer.emoji;

            document.getElementById('cheerMessage').textContent = cheer.msg;

           

            // حركة نبض للبانر

            banner.classList.remove('scale-[1.01]', 'bg-amber-50');

            void banner.offsetWidth; // trigger reflow

            banner.classList.add('scale-[1.01]', 'bg-amber-50');

            setTimeout(() => banner.classList.remove('scale-[1.01]', 'bg-amber-50'), 600);

 

            // إضافة للسجل المباشر

            const nowTime = new Date().toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' });

            let logText = `${rep.avatar} ${rep.name} سجل حجز جديد! وصل لـ ${rep.count}/${rep.target} عميل`;

            let logType = 'booking';

 

            if (isTargetReached) {

                logText = `🏆 إنجاز تاريخي! ${rep.name} قفل التارجت الكامل (${rep.target}/${rep.target})! 🎉`;

                logType = 'milestone';

            } else if (isMilestoneUp) {

                logText = `${newMilestone.icon} ترقية! ${rep.name} فتح وسام "${newMilestone.title}" (${rep.count}/${rep.target})`;

                logType = 'milestone';

            }

 

            appState.logs.unshift({

                id: Date.now(),

                text: logText,

                time: nowTime,

                type: logType

            });

 

            // الاحتفاظ بآخر 20 حدث فقط

            if (appState.logs.length > 20) {

                appState.logs.pop();

            }

 

            // تشغيل المؤثرات الصوتية والبصرية

            playCelebrationSound(isMilestoneUp, isTargetReached);

            triggerConfetti(isTargetReached || isMilestoneUp);

 

            saveState();

            renderAll();

        }

 

        function removeBooking(repId) {

            const rep = appState.reps.find(r => r.id === repId);

            if (!rep || rep.count <= 0) return;

 

            rep.count -= 1;

            const nowTime = new Date().toLocaleTimeString('ar-EG', { hour: '2-digit', minute: '2-digit' });

            appState.logs.unshift({

                id: Date.now(),

                text: `↩️ تم التراجع عن حجز لـ ${rep.name} (الرصيد الحالي: ${rep.count}/${rep.target})`,

                time: nowTime,

                type: 'undo'

            });

 

            saveState();

            renderAll();

        }

 

        function triggerConfetti(bigBurst = false) {

            if (typeof confetti !== 'function') return;

            if (bigBurst) {

                confetti({

                    particleCount: 120,

                    spread: 80,

                    origin: { y: 0.6 },

                    colors: ['#0ea5e9', '#f59e0b', '#10b981', '#f43f5e']

                });

            } else {

                confetti({

                    particleCount: 45,

                    spread: 55,

                    origin: { y: 0.7 },

                    colors: ['#0ea5e9', '#fbbf24', '#38bdf8']

                });

            }

        }

            const totalCount = appState.reps.reduce((sum, r) => sum + r.count, 0);

            const totalTarget = appState.reps.reduce((sum, r) => sum + r.target, 0);

            const percent = totalTarget > 0 ? Math.min(100, Math.round((totalCount / totalTarget) * 100)) : 0;

 

            document.getElementById('totalCompanyCount').textContent = totalCount;

            document.querySelector('#totalCompanyCount + span').textContent = `/ ${totalTarget} عميل`;

            document.getElementById('totalCompanyPercent').textContent = `${percent}%`;

            document.getElementById('totalCompanyBar').style.width = `${percent}%`;

        }

        function selectAvatar(av) {

            document.getElementById('editRepAvatar').value = av;

            document.querySelectorAll('.avatar-opt').forEach(btn => {

                    btn.className = 'avatar-opt text-2xl p-2 rounded-xl border-2 transition cursor-pointer border-sky-500 bg-sky-50 scale-105';

                }

        }

        function saveRepEdit(e) {

            e.preventDefault();

            const id = parseInt(document.getElementById('editRepId').value);

            const name = document.getElementById('editRepName').value.trim();

            const target = parseInt(document.getElementById('editRepTarget').value);

            const avatar = document.getElementById('editRepAvatar').value;

 

            const rep = appState.reps.find(r => r.id === id);

            if (rep && name && target > 0) {

                rep.name = name;

                rep.target = target;

                rep.avatar = avatar;

                saveState();

                renderAll();

            }

            closeEditModal();

        }
